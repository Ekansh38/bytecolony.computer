BreadcrumbList mirroring the visible breadcrumbs */}} {{- $site := "https://bytecolony.computer" -}} {{- $person := dict "@type" "Person" "@id" (printf "%s/#person" $site) "name" "Ekansh Goenka" "url" $site "name" "ByteColony" "inLanguage" "en" "publisher" (dict "@id" (printf "%s/#person" $site)) -}} {{- $graph := slice $person $website -}} {{- if and .IsPage (or (eq .Section "writing") (eq .Section "projects")) -}} {{- $desc := "" -}} {{- with .Params.description }}{{ $desc = . }}{{ else }}{{ $desc = .Summary | plainify | truncate 200 }}{{ end -}} {{- $img := printf "%s/images/og.png" $site -}} {{- $article := dict "@type" "TechArticle" "@id" (printf "%s#article" .Permalink) "headline" .Title "name" .Title "description" $desc "url" .Permalink "datePublished" (.Date.Format "2006-01-02") "dateModified" (.Lastmod.Format "2006-01-02") "wordCount" .WordCount "image" $img "inLanguage" "en" "isAccessibleForFree" true "author" (dict "@id" (printf "%s/#person" $site)) "alternativeHeadline" .) }}{{ end -}} {{- with .Params.about }}{{ $article = merge $article (dict "about" .) }}{{ end -}} {{- with .Params.teaches }}{{ $article = merge $article (dict "teaches" .)
